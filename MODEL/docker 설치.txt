docker 설치

yum install yum-utils 	// 필요한 패키지 설치
sudo yum-config-manager --add-repo https://download.docker.com/linux/centos/docker-ce.repo	// repo 등록
dnf remove podman*	// podman 삭제
dnf remove containers-common-1-43.module+el8.7.0+1078+e72fcd4f.x86_64	// containers 관련 rpm 삭제
dnf install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin	// docker 설치
systemctl start docker	// docker 서비스 시작
systemctl enable docker	// docker 서비스 등록
docker run -d -p 80:80 --name docker-test-server nginx	// docker 시작
docker ps	// 실행중인 도커 목록 확인
docker stop docker-test-server	// 도커 중지
docker start docker-test-server	// 도커 실행
docker commit apache-webserver apache-server-copy-image:latest	// 도커내용 이미지로 생성
docker run -d -p 82:80 --name apache-webserver2 apache-server-copy-image:latest	// 이미지 파일로 도커 생성
docker exec -i apache-webserver sh -c "echo '<h1>It works! 81</h1>' > /usr/local/apache2/htdocs/index.html" 2>/dev/null	// docker apache 내용 수정
docker exec -i apache-webserver2 sh -c "echo '<h1>It works! 82</h1>' > /usr/local/apache2/htdocs/index.html" 2>/dev/null	// docker apache 내용 수정

nginx 설정
cat << 'EOF' > /root/nginx.conf
upstream backend {
    server 192.168.1.50:81;
    server 192.168.1.50:82;
}

server {
    listen 80;
    server_name localhost;

    location / {
        proxy_pass http://backend;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}
EOF

docker stop docker-test-server	// docker 중지
docker rm docker-test-server		// docker 삭제
docker run -d -p 80:80 --name docker-test-server -v /root/nginx.conf:/etc/nginx/conf.d/default.conf:ro nginx	// NGINX 새로 DOCKER 생성
